function (triggerId, reason)
    debug("newChar system initiated. Reason: " .. (reason or "unknown"))
    
    ---------------------------------------
    -- ★ 밸런싱 상수 정의 (Balancing Constants)
    -- 신규 캐릭터 생성 시스템의 핵심 수치를 조정하려면 이 섹션의 값을 수정하세요.
    ---------------------------------------

    -- [초상화 생성 프롬프트 CONSTANT]
    local PORTRAIT_PROMPT_PREFIX = "a hero portraiticon of a dark fantasy game, portrait, face focus, high quality, detailed, 8k, dark fantasy mood"

    -- [HP/SP 랜덤 변동폭]
    local HP_SP_RANDOM_MIN      = 90    -- HP/SP 최종값 최소 배율 (%). 최종 HP/SP는 기본값의 이 % ~ MAX % 사이에서 결정.
    local HP_SP_RANDOM_MAX      = 110   -- HP/SP 최종값 최대 배율 (%).

    -- [신체특성 부여 확률] (순차 적용)
    local TRAIT_PROB_BODY_1     = 0.6   -- 첫 번째 신체특성 부여 확률 (60%).
    local TRAIT_PROB_BODY_2     = 0.3   -- 두 번째 신체특성 부여 확률 (30%). 첫 번째 획득 시에만 판정.
    local TRAIT_PROB_BODY_3     = 0.15  -- 세 번째 신체특성 부여 확률 (15%). 두 번째 획득 시에만 판정.

    -- [성적취향 부여 확률]
    local TRAIT_PROB_SEXUAL     = 0.1   -- 성적취향 특성 부여 확률 (10%). 10% 확률로 성적취향 1개 부여.

    -- [성격 부여 확률]
    local TRAIT_PROB_EXTRA_PERS = 0.5   -- 추가 성격 부여 확률 (50%). 기본 1개 + 이 확률로 1개 추가.

    ---------------------------------------


    -- 1. 설정 및 데이터 로드
    local raceDB = getDB(triggerId, "race.db")
    local classDB = getDB(triggerId, "class.db")
    local traitDB = getDB(triggerId, "trait.db")
    local expTable = getDB(triggerId, "EXPtable.db") 
    
    local dungeonLv = tonumber(getChatVar(triggerId, "dungeonLv") or 1)
    local ageLimit = tonumber(getChatVar(triggerId, "ageLimit") or 12)
    local raceLimit = getChatVar(triggerId, "raceLimit") or "0" 
    local maleLimit = getChatVar(triggerId, "maleLimit") or "0"
    local futaLimit = getChatVar(triggerId, "futaLimit") or "0"
    local heroLvLimit = tonumber(getChatVar(triggerId, "heroLvLimit") or 5)
    
    -- 2. 기본 골격 생성 (RNG)
    math.randomseed(os.time())
    
    local level = math.max(1, dungeonLv + math.random(-2, heroLvLimit))
    
    -- 종족 결정
    local raceListKey = json.decode(getChatVar(triggerId, "raceList"))
    local raceKey = ""
    local raceData = {}
    
    if raceLimit == "1" then -- 인간만
        raceKey = "human"
    else
        raceKey = raceListKey[math.random(#raceListKey)]
    end
    raceData = raceDB[raceKey] -- 객체 구조
    -- 혹시 데이터 없으면 인간으로 폴백
    if not raceData then raceKey = "human"; raceData = raceDB["human"] end


    -- 클래스 결정
    local classListKey = json.decode(getChatVar(triggerId, "classList"))
    local classKey = classListKey[math.random(#classListKey)]
    local classData = classDB[classKey]

    -- 성별 결정
    local genderPool = {"여성", "여성", "여성"} 
    if maleLimit == "1" then table.insert(genderPool, "남성") end
    if futaLimit == "1" then table.insert(genderPool, "후타나리") end
    local gender = genderPool[math.random(#genderPool)]
    
    -- 나이 결정
    local ageBase = math.max(ageLimit, 8)
    local lifeSpanMult = raceData.life_mod or 1.0 -- 필드명 변경 반영
    local age = math.floor((ageBase + math.random(0, 10)) * lifeSpanMult)
    if age < ageLimit then age = ageLimit end
    
    -- 3. 스탯 계산 (1레벨 기본 스탯 설정 후 LvUp.sys로 성장)
    local raceBaseHP = raceData.base_hp or 100
    local raceBaseSP = raceData.base_sp or 100
    
    local raceAtkBonus = raceData.bonus_atk or 0
    local raceDefBonus = raceData.bonus_def or 0
    
    -- 1레벨 기본 스탯 (종족 기본치 + 종족 보너스 + 1레벨 클래스 성장치)
    local maxHP = math.floor(raceBaseHP * (math.random(HP_SP_RANDOM_MIN, HP_SP_RANDOM_MAX) * 0.01))
    local maxSP = math.floor(raceBaseSP * (math.random(HP_SP_RANDOM_MIN, HP_SP_RANDOM_MAX) * 0.01))
    
    -- 클래스 초기 스탯 (1레벨분)
    local classAtkMin = classData.atk_min or 3
    local classAtkMax = classData.atk_max or 6
    local classDef = classData.def_inc or 5
    
    local atkMin = raceAtkBonus + math.random(1, classAtkMin)
    local atkMax = raceAtkBonus + math.random(1, classAtkMax)
    
    -- [안전장치] min > max 역전 시 스왑
    if atkMax < atkMin then
        local temp = atkMin
        atkMin = atkMax
        atkMax = temp
    end
    
    local def = raceDefBonus + math.random(1, classDef)


    -- 신체 사이즈
    local heightBase = raceData.height or 160
    local height = math.floor(heightBase + math.random(-10, 10))
    local weight = math.floor((height - 100) * 0.9 * (math.random(90, 110)/100))
    
    local bust = "평범"
    local nipple = "갈색"
    local penis = ""
    local pubes = "보통"
    
    if gender ~= "남성" then
        local bustSizes = {"절벽", "빈유", "평범", "거유", "폭유"}
        bust = bustSizes[math.random(#bustSizes)]
    end
    
    if gender ~= "여성" then
        local penisSizes = {"실좆", "작음", "평범", "대물", "말자지"}
        penis = penisSizes[math.random(#penisSizes)]
    end
    
    -- 4. 특성(Trait) 부여 (한글 이름 직접 선택)
    local personalityList = json.decode(getChatVar(triggerId, "성격List"))
    local bodyTraitList = json.decode(getChatVar(triggerId, "신체특성List"))
    
    local assignedTraits = {} -- 한글 이름 저장
    
    -- 특성 추가 함수 (중복 체크)
    local function addTrait(list)
        if not list or #list == 0 then return false end
        local name = list[math.random(#list)]
        
        local dup = false
        for _, n in ipairs(assignedTraits) do if n == name then dup = true; break end end
        if not dup then 
            table.insert(assignedTraits, name) 
            return true
        end
        return false
    end
    
    -- 성격 1~2개 
    local pAdded = 0
    local retry = 10
    while pAdded < 1 and retry > 0 do
        if addTrait(personalityList) then pAdded = pAdded + 1 end
        retry = retry - 1
    end
    if math.random() > TRAIT_PROB_EXTRA_PERS then addTrait(personalityList) end
    
    -- 신체특성 추가 (확률 순차 적용, 최대 3개)
    if math.random() <= TRAIT_PROB_BODY_1 then
        addTrait(bodyTraitList)
        if math.random() <= TRAIT_PROB_BODY_2 then
            addTrait(bodyTraitList)
            if math.random() <= TRAIT_PROB_BODY_3 then
                addTrait(bodyTraitList)
            end
        end
    end

    -- 성적취향 추가 (확률 기반 1개)
    local sexualPrefList = json.decode(getChatVar(triggerId, "성적취향List"))
    if math.random() <= TRAIT_PROB_SEXUAL then
        addTrait(sexualPrefList)
    end
    
    
    -- 5. 스켈레톤 데이터 구성 (1레벨 기본 스탯으로 시작)
    local skeleton = {
        ["종족"] = raceData.name, -- "인간"
        ["성별"] = gender,
        ["나이"] = tostring(age),
        ["클래스"] = classData.name, -- "기사"
        ["레벨"] = "1", -- 1레벨에서 시작하여 LvUp.sys로 성장
        ["경험치"] = "0",
        
        ["체력"] = tostring(maxHP),
        ["최대체력"] = tostring(maxHP),
        ["기력"] = tostring(maxSP),
        ["최대기력"] = tostring(maxSP),
        ["공격력"] = atkMin .. "~" .. atkMax,
        ["방어력"] = def,
        ["레이팅"] = 0, -- LvUp.sys에서 계산
        
        ["자궁정액"]="0",
        ["애널정액"]="0",
        ["condition"] = "REST",

        ["키"] = tostring(height),
        ["몸무게"] = tostring(weight),
        ["가슴"] = bust,
        ["유두"] = nipple,
        ["음모"] = pubes,
        ["페니스"] = penis,
        ["문신"]="",
        ["피어싱"]="",

        ["무기"] = classData.weapon, -- 한글
        ["방어구"] = classData.armor,
        ["장신구"]="없음",

        ["성격"] = {},
        ["신체특성"] = {},
        ["성적취향"] = {},
        ["칭호"] = "미함락",
        
        ["첫키스 상대"]="미경험", ["첫경험 상대"]="미경험", ["이상경험 기록"]={}
    }

    -- 특성 주입: UI 호환을 위해 최상위에 1로 저장
    for _, traitName in ipairs(assignedTraits) do
        skeleton[traitName] = 1
    end
    
    -- 6. LvUp.sys를 사용하여 목표 레벨까지 성장 (1레벨에서 시작)
    if level > 1 then
        skeleton, _, _ = sysFunction(triggerId, "LvUp.sys", "execute", skeleton, level - 1, true)
        debug(string.format("캐릭터 레벨업 완료: Lv.%s, ATK:%s, DEF:%s, Rating:%s", 
            skeleton["레벨"], skeleton["공격력"], skeleton["방어력"], skeleton["레이팅"]))
    else
        -- 1레벨이면 Rating만 계산
        skeleton = sysFunction(triggerId, "LvUp.sys", "calcRating", skeleton)
    end
    
    debug("기초데이터", json.encode(skeleton))
    
    -- 7. LLM 요청 (설정 생성)
    local prompt = {
        promptBuild("system", getLoreBookContent(triggerId, "newChar.pt")),
        promptBuild("system", "existing names: "..getChatVar(triggerId, "chars")),
        promptBuild("user", json.encode(skeleton))
    }
    
    debug("신규 용사 생성 요청...")
    
    local response = LLM(triggerId, prompt, "newChar")
    if not response.success then
        alertError(triggerId, "용사 생성 실패 (LLM Error)")
        return
    end
    
    local content = response.result
    content = content:gsub("<Thoughts>.-</Thoughts>", "") 
    local narrative = json.decode(content:match("({.*})"))
    
    -- 8. 데이터 병합 & 이미지 생성
    for k, v in pairs(narrative) do
        skeleton[k] = v
    end

    -- [초상화 생성 로직]
    local appearanceDesc = skeleton["외형"] or skeleton["appearance"] or (skeleton["머리스타일"] and (skeleton["머리스타일"].." hair")) or ""
    
    -- 간단한 번역/매핑
    local genderStr = (skeleton["성별"] == "남성" and "male") or (skeleton["성별"] == "후타나리" and "futanari") or "female"
    local raceStr = skeleton["종족"] or "human"
    
    local fullPrompt = PORTRAIT_PROMPT_PREFIX .. ", " .. genderStr .. ", " .. raceStr .. ", " .. appearanceDesc

    debug("초상화 생성 시도: " .. fullPrompt)
    local imgResult = generateImage(triggerId, fullPrompt):await()
    
    if imgResult then
        local cleanImg = imgResult:gsub("{{inlay:", "")
        -- [JSON Parsing Fix] Split ':' into two fields
        skeleton["portrait_pre"] = "{{inlay:"
        skeleton["portrait_append"] = cleanImg
        skeleton["portrait"] = nil -- 기존 필드 제거
        debug("초상화 저장 완료")
    else
        debug("초상화 생성 실패")
    end


    -- DB에 없는 필드 0으로 초기화
    local extraDBs = {"mark.db", "abl.db", "exp.db", "stat.db"}
    for _, dbName in ipairs(extraDBs) do
        local dbContent = getDB(triggerId, dbName) -- 객체
        for key, val in pairs(dbContent) do
            -- 여기서도 한글 매핑 필요. charInfo.html 등에서 한글 키로 접근할 것임.
            if val.name then
                if not skeleton[val.name] then skeleton[val.name] = "0" end
            else
                if not skeleton[key] then skeleton[key] = "0" end
            end
        end
    end

    -- Trace 초기화
    local allTrace = getState(triggerId, "trace") or {}
    local traceDB = getDB(triggerId, "trace.db")
    allTrace[skeleton["이름"]] = {}
    
    for k, v in pairs(traceDB) do
        if v.name then
            allTrace[skeleton["이름"]][v.name] = 0
        end
    end
    setState(triggerId, "trace", allTrace)
    
    -- 캐릭터 리스트 업데이트
    local chars = json.decode(getChatVar(triggerId, "chars"))
    table.insert(chars, skeleton["이름"])
    setChatVar(triggerId, "chars", json.encode(chars))
    
    -- 로어북 저장
    upsertLocalLoreBook(triggerId, skeleton["이름"], json.encode(skeleton), DEFAULT_LORE_OPTION)
    
    debug("신규 용사 생성 완료", skeleton["이름"])
    
    if getState(triggerId, "screen") == "charList" then
        sysFunction(triggerId, "char.sys", "list")
    end
end

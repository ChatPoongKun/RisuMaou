-- char.sys.txt
-- 캐릭터 관리 통합 모듈
-- 사용법: sysFunction(triggerId, "char.sys", "card", name)
--         sysFunction(triggerId, "char.sys", "list")

(function()
    ---------------------------------------
    -- 로컬 함수 정의
    ---------------------------------------

    ---------------------------------------
    -- 디스패처 테이블 및 로컬 변수
    ---------------------------------------
    local funcs = {}

    -- 캐릭터 카드 선택 처리
    function funcs.card(triggerId, name)
        debug(string.format("[Char:Select] Target: %s, Screen: %s", name, getState(triggerId, "screen")))
        local screen = getState(triggerId, "screen")
        
        -- [Migration] Trace Migration & UI Refresh (Shared Logic)
        local traceDB = getDB(triggerId, "trace.db")
        local traceList = {}
        for k, v in pairs(traceDB) do
            if v.name then table.insert(traceList, v.name) end
        end
        table.sort(traceList)
        setChatVar(triggerId, "traceList", json.encode(traceList))

        -- 데이터 로드 및 마이그레이션
        local allTraces = getState(triggerId, "trace")
        if not allTraces[name] then allTraces[name] = {} end
        local targetTrace = allTraces[name]
        
        local oldToNew = {
            ["C감각"] = "C감각의 흔적", ["B감각"] = "B감각의 흔적", ["V감각"] = "V감각의 흔적", ["A감각"] = "A감각의 흔적",
            ["U감각"] = "U감각의 흔적", ["애정"] = "애정의 흔적", ["발정"] = "발정의 흔적", ["순종"] = "순종의 흔적",
            ["복종"] = "복종의 흔적", ["기술"] = "기술의 흔적", ["굴욕"] = "굴욕의 흔적", ["상처"] = "상처의 흔적",
            ["불안"] = "불안의 흔적", ["부정"] = "부정의 흔적"
        }
        local migrated = false
        for old, new in pairs(oldToNew) do
            if targetTrace[old] and (tonumber(targetTrace[old]) or 0) > 0 then
                targetTrace[new] = (targetTrace[new] or 0) + tonumber(targetTrace[old])
                targetTrace[old] = nil -- Remove old key
                migrated = true
            end
        end
        
        if migrated then
            allTraces[name] = targetTrace
            setState(triggerId, "trace", allTraces)
            debug("Trace Migrated in char.sys for " .. name)
        end

        if screen == "charList" then --캐릭터 조회화면
            if name == getPersonaName(triggerId) then
                name = "user"
            end
            local target = json.decode(getLoreBookContent(triggerId, name))
            setState(triggerId, "target", target)
            stateToVar(triggerId, "target", target)
            stateToVar(triggerId, "targetTrace", targetTrace)
            setChatVar(triggerId, "category", "appearance")
            screenChange(triggerId, "charInfo") --캐릭터 상세 정보 화면으로 이동
        elseif screen == "disposal" then -- 포로 처분 화면
            -- 처분 방식 선택 실행
            sysFunction(triggerId, "disposal.sys", "selectMethod", name)
        elseif screen == "preTrain" then        --조교 대상 화면
            sysFunction(triggerId, "train.sys", "prepare", name)
            stateToVar(triggerId, "targetTrace", targetTrace)
            screenChange(triggerId, "train") --조교 화면으로 이동
        end
    end

    -- 버튼 클릭 핸들러
    -- data format: "char.sys_card_name"
    function funcs.click(triggerId, data)
        -- sys파일명_card_name 형태에서 name 추출
        local name = string.match(data, "card_(.*)")
        if name then
            funcs.card(triggerId, name)
        else
            debug("char.sys: 잘못된 데이터 포맷 - " .. tostring(data))
        end
    end

    -- 캐릭터 목록 생성
    function funcs.list(triggerId)
        local chars = json.decode(getChatVar(triggerId, "chars"))
        local html = ""
        local screen = getState(triggerId, "screen")
        local sortMode = getChatVar(triggerId, "category") or "index" -- 기본값 index

        -- 1. 데이터 수집 및 전처리
        local charDataList = {}
        -- [Refactoring] 필터링 로직 함수화 (확장성을 위해)
        local function checkFilter(scr, c)
            local cond = c["condition"]
            if scr == "preTrain" then
                -- preTrain: 오직 '조교가능' 상태인 캐릭터만 표시
                return cond == "TRAINABLE"
            elseif scr == "charInfo" then
                -- charInfo: 사망한 캐릭터도 포함하여 모두 표시
                return true
            else
                -- 기본값: 사망하지 않은 캐릭터만 표시
                return cond ~= "DEAD"
            end
        end

        for i, name in ipairs(chars) do
            -- preTrain 화면에서는 user 제외 (user는 chars 리스트에 보통 없지만 체크 필요시 추가)
            local char = json.decode(getLoreBookContent(triggerId, name))

            if checkFilter(screen, char) then
                -- 정렬 및 표시에 필요한 데이터 미리 추출
                local data = {
                    index = i,
                    name = name,
                    raw = char,
                    lv = tonumber(char["레벨"] or 0),
                    hp = tonumber(char["체력"] or 0),
                    hpMax = tonumber(char["최대체력"] or 0),
                    st = tonumber(char["기력"] or 0),
                    stMax = tonumber(char["최대기력"] or 0),
                    atk = char["공격력"] or "0",
                    def = char["방어력"] or "0",
                    rating = tonumber(char["레이팅"] or 0),
                    condition = char["condition"] or "",
                    title = char["칭호"] or ""
                }
                table.insert(charDataList, data)
                
                -- Condition DB Load (Lazy) (Cached outside or loaded every list call is fine for now)
                local condDB = getDB(triggerId, "condition.db")
                if condDB[data.condition] and condDB[data.condition].name then
                   data.conditionDisplay = condDB[data.condition].name
                else
                   data.conditionDisplay = data.condition -- Fallback (e.g. if already Korean or unknown)
                end
            end
        end

        -- 2. 정렬 로직 수행
        if sortMode == "level" then
            table.sort(charDataList, function(a, b) return a.lv > b.lv end)         -- 레벨 내림차순
        elseif sortMode == "name" then
            table.sort(charDataList, function(a, b) return a.name < b.name end)     -- 이름 오름차순
        elseif sortMode == "hp" then
            table.sort(charDataList, function(a, b) return a.hp > b.hp end)         -- 체력 비율 오름차순
        elseif sortMode == "rating" then
            table.sort(charDataList, function(a, b) return a.rating > b.rating end) -- 레이팅 내림차순
        else
            -- index 정렬 (기본)
            table.sort(charDataList, function(a, b) return a.index < b.index end)
        end

        -- 3. HTML 생성
        for _, data in ipairs(charDataList) do
            local name = data.name

            -- 체력/기력 비율 계산
            local hpRatio = 0
            if data.hpMax > 0 then hpRatio = (data.hp / data.hpMax) * 100 end

            local stRatio = 0
            if data.stMax > 0 then stRatio = (data.st / data.stMax) * 100 end

            -- 칭호 태그 스타일 결정
            local tagClass = ""
            local tagHtml = ""
            local title = data.title

            if title ~= "" then
                if title == "음란" then
                    tagClass = "tag-lewd"
                elseif title == "연모" then
                    tagClass = "tag-love"
                elseif title == "암캐" then
                    tagClass = "tag-puppy"
                elseif title == "붕괴" then
                    tagClass = "tag-destroyed"
                elseif title == "광기" then
                    tagClass = "tag-mad"
                else
                    tagClass = "tag-unconquered" -- 기본값
                end
                tagHtml = [[<span class="tag ]] .. tagClass .. [["> 〈]] .. title .. [[〉</span>]]
            end

            -- 상태에 따른 클래스 추가 (사망 등)
            local cardClass = "character-card"
            if data.condition == "DEAD" or data.condition == "사망" then cardClass = cardClass .. " 사망" end
            if name == "user" then name = "{{" .. name .. "}}" end

            -- HTML 조립
            -- HTML 조립
            local portraitHtml = ""
            local p_pre = data.raw.portrait_pre
            local p_app = data.raw.portrait_append
            local portraitContent = ""

            if p_pre and p_app then
                portraitContent = p_pre .. p_app
            elseif data.raw.portrait and data.raw.portrait ~= "" then
                -- 기존 방식 (Fallback)
                local val = data.raw.portrait
                -- 일반 경로 텍스트면 {{raw::...}} 로 감싸고 img 태그 추가
                if not val:match("^{{") and not val:match("^<img") then
                     val = "{{raw::" .. val .. "}}"
                end
                
                -- 이미지가 아니라 inlay등의 매크로라면 그대로, 아니면 img 태그로
                if val:match("{{inlay") or val:match("^<div") or val:match("^<img") then
                     portraitContent = val
                else
                     portraitContent = [[<img src="]] .. val .. [[" style="width:100%; height:100%; object-fit:cover;">]]
                end
            end
            
            if portraitContent ~= "" then
                 portraitHtml = [[<div style="width:50px; height:50px; overflow:hidden; border-radius:4px; margin-right:8px; flex-shrink:0;">]] .. portraitContent .. [[</div>]]
            end

            local card = [[
<div class="]] .. cardClass .. [[" risu-btn="char.sys_card_]] .. name .. [[">
<div class="char-info">
]] .. portraitHtml .. [[
<div style="flex:1;">
<p class="char-name-line">
<span class="char-condition">[]] ..
                data.conditionDisplay .. [[] </span>]] .. name .. [[<span class="char-level"> Lv.]] .. data.lv .. [[ </span>
]] .. tagHtml .. [[
</p>
<span class="char-tags">
AT ]] .. data.atk .. [[ / ]] .. data.def .. [[ / RT ]] .. data.rating .. [[
</span>
</div>
</div>
<div class="char-status-bars">
<div class="x-risu-bar-container">
<span class="bar-label">체력:</span>
<div class="progress-bar">
<div class="x-risu-progress-bar-fill x-risu-hp" style="width: ]] .. hpRatio .. [[%;"></div>
</div>
<span class="bar-value">]] .. data.hp .. [[</span>
</div>
<div class="x-risu-bar-container">
<span class="bar-label">기력:</span>
<div class="progress-bar">
<div class="x-risu-progress-bar-fill x-risu-st" style="width: ]] .. stRatio .. [[%;"></div>
</div>
<span class="bar-value">]] .. data.st .. [[</span>
</div>
</div>
</div>]]

            html = html .. card
        end

        if html == "" then
            html = "<div class='status-message'><p>표시할 캐릭터가 없습니다.</p></div>"
        end

        setChatVar(triggerId, "charList", html)
    end

    ---------------------------------------
    -- 메인 함수 (외부에서 호출됨)
    ---------------------------------------
    return function(triggerId, subFunc, ...)
        if funcs[subFunc] then
            return funcs[subFunc](triggerId, ...)
        else
            debug("char.sys: 알 수 없는 함수 - " .. tostring(subFunc))
        end
    end
end)()

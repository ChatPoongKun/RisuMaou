function (triggerId)
    debug("turnEnd 실행")
    
    --HP/SP 회복 처리 함수
    --isNight: true면 밤(높은 회복률), false면 낮(낮은 회복률)
    local function processRecovery(triggerId, charName, isNight)
        local char = json.decode(getLoreBookContent(triggerId, charName))
        
        --기본 회복률 설정
        local hpGen, spGen
        if isNight then
            hpGen = 0.1
            spGen = 0.15
        else
            hpGen = 0.05
            spGen = 0.1
        end
        
        --빠른회복 미보유 또는 실신 시 회복률 감소
        if char["빠른회복"] == 0 or char["condition"] == "실신" then
            hpGen = hpGen - 0.05
            if not isNight then
                hpGen = 0  --낮에는 회복 없음
            end
        end
        
        --실신 시 HP 회복 50% 감소
        if char["condition"] == "실신" then
            hpGen = hpGen * 0.5
        end
        
        --실제 회복 계산
        local maxHP = tonumber(char["최대체력"])
        local maxSP = tonumber(char["최대기력"])
        local currentHP = tonumber(char["체력"])
        local currentSP = tonumber(char["기력"])
        
        char["체력"] = intForm(math.min(currentHP + math.floor(maxHP * hpGen), maxHP))
        char["기력"] = intForm(math.min(currentSP + math.floor(maxSP * spGen), maxSP))
        
        --기력이 90% 이상 회복되면 실신에서 조교가능으로 변경
        if tonumber(char["기력"]) > maxSP * 0.9 and char["condition"] == "실신" then
            char["condition"] = "조교가능"
        end
        
        --로어북에 저장
        upsertLocalLoreBook(triggerId, charName, json.encode(char), DEFAULT_LORE_OPTION)
        
        return char
    end
    
    --시간 정보 호출
    local day = getChatVar(triggerId, "day")
    local ampm = getChatVar(triggerId, "ampm")
    local chars = json.decode(getChatVar(triggerId, "chars"))
    local isNight = (ampm == "1")
    
    -- [침공 이벤트 트리거] - 알림보다 먼저 체크
    -- 조건: 3일차 이상, 밤(isNight=true), 50% 확률
    local invasionTriggered = false
    if isNight and tonumber(day) >= 3 and math.random() < 0.5 then
        invasionTriggered = true
        
        -- 즉시 배틀화면으로 전환 (로딩 상태 표시)
        setChatVar(triggerId, "battleLoadingState", "침공 용사 파악중...")
        setChatVar(triggerId, "battleLogsHtml", "") -- 초기화
        setChatVar(triggerId, "battleNarration", "")
        setChatVar(triggerId, "defenderImage", "")
        setChatVar(triggerId, "battleDungeonHtml", "")
        screenChange(triggerId, "battle")
        reloadDisplay(triggerId) -- 화면 갱신
        
        -- 전투 시뮬레이션 실행
        -- setChatVar(triggerId, "battleLoadingState", "전투 시뮬레이션 중...")
        -- reloadDisplay(triggerId)
        local summary = sysFunction(triggerId, "battle.sys", "processInvasion")
        debug("전투 결과 수신 완료")
        
        if summary and summary.logs and #summary.logs > 0 then
            local turnInterval = 1.2
            
            -- [No-JS] Dungeon Status HTML 가져오기 (Battle Status with Timeline)
            -- dungeon.sys의 getBattleStatusHtml이 이제 timeline을 인자로 받음
            -- setChatVar(triggerId, "battleLoadingState", "던전 상태 생성중...")
            -- reloadDisplay(triggerId)
            local dungeonStatusHtml = sysFunction(triggerId, "dungeon.sys", "getBattleStatusHtml", summary.clearedSlots, summary.timeline, turnInterval)
            setChatVar(triggerId, "battleDungeonHtml", dungeonStatusHtml)
            debug("던전 상태 HTML 생성 완료")

            -- [No-JS] HTML 로그 생성 (Turn-based Batch 처리) & Defender Image Stack 생성
            local logLines = {}
            local currentTurnLines = {}
            local turnDelay = 0.0
            
            local lastTurnNum = -1
            
            -- Defender Stack Logic
            local defenderStack = {}
            -- 기본 이미지 (마왕) 삭제 - 몬스터만 표시하거나 비워둠
            
            -- 타임라인 기반 Defender Image 생성
            -- summary.timeline: { {startTurnIndex=0, defender={type="monster", key="..."}}, ... }
            -- 타임라인 기반 Defender Image 생성 (역순 적재: 나중 턴이 아래깔림, 앞 턴이 위에 덮임)
            -- [Boss(Base), ..., Mon2(Timed), Mon1(Timed)] 순서로 DOM에 추가됨 -> Mon1이 맨 위
            if summary.timeline then
                -- 역순 순회
                for i = #summary.timeline, 1, -1 do
                    local event = summary.timeline[i]
                    local dKey = event.defender and event.defender.key
                    local dType = event.defender and event.defender.type
                    
                    if dKey and dType ~= "none" then
                        -- Duration 계산 (다음 이벤트까지의 시간)
                        -- 현재 i번째 이벤트의 duration은 정순일 때의 [i+1] - [i]
                        -- 정순 인덱스 i 기준으로 nextEvent는 timeline[i+1]
                        local duration = turnInterval * 5 -- 기본값
                        local nextEvent = summary.timeline[i + 1]
                        if nextEvent and nextEvent.startTurnIndex then
                            duration = (nextEvent.startTurnIndex - event.startTurnIndex) * turnInterval
                        end
                        
                        -- 퇴장 시간(Delay) = (시작 턴 + 지속 턴) * 턴시간
                        local exitTime = (event.startTurnIndex * turnInterval) + (duration * 0.9) -- 겹침 방지 위해 살짝 일찍 컷
                        
                        local src = ""
                        if dType == "monster" then
                            src = "{{raw::" .. dKey .. "}}"
                        elseif dType == "boss" or dKey == "user" then
                             src = "{{source::user}}"
                        else
                             src = "{{raw::" .. dKey .. "}}" 
                        end
                        
                        -- 마지막 요소(타임라인상 가장 뒤 = 보스)는 사라지지 않는 Base
                        if i == #summary.timeline then
                            table.insert(defenderStack, string.format('<img src="%s" class="defender-img base">', src))
                        else
                            -- 그 외 요소는 시간이 되면 사라짐 (Fade Out)
                            -- animation: name duration timing-function delay fill-mode
                            table.insert(defenderStack, string.format(
                                '<img src="%s" class="defender-img timed" style="animation: peel-off 0.5s ease-out %.2fs forwards;">', 
                                src, exitTime
                            ))
                        end
                    end
                end
            end
            
            local defenderHtml = table.concat(defenderStack, "\n")
            setChatVar(triggerId, "defenderImage", defenderHtml) -- 변수명 재사용 (HTML 삽입)

            -- 로그 처리: 턴 번호가 바뀔 때마다 딜레이를 누적하여 순차 출력
            -- (예: 몬스터1의 Turn1,2,3 -> 몬스터2의 Turn1,2,3 순서로)
            for i, line in ipairs(summary.logs) do
                local turnNum = tonumber(line:match("%[Turn (%d+)%]")) or lastTurnNum
                
                if turnNum ~= lastTurnNum then
                    -- 이전 턴의 로그들을 현재 딜레이로 출력
                    if #currentTurnLines > 0 then
                        for _, l in ipairs(currentTurnLines) do
                            table.insert(logLines, string.format('<div class="%s" style="animation-delay: %.2fs">%s</div>', l.class, turnDelay, l.text))
                        end
                        -- 중요: 누적 딜레이 사용 (절대값 X)
                        turnDelay = turnDelay + turnInterval
                        currentTurnLines = {}
                    end
                    lastTurnNum = turnNum
                end
                
                local class = "log-line"
                if string.find(line, "처치") or string.find(line, "승리") or string.find(line, "제압") then
                    class = class .. " victory"
                elseif string.find(line, "피해") or string.find(line, "쓰러졌") then
                    class = class .. " danger"
                elseif string.find(line, "Turn") then
                    class = class .. " highlight"
                end
                
                table.insert(currentTurnLines, {text=line, class=class})
            end
            
            if #currentTurnLines > 0 then
                 -- 마지막 턴 로그 처리
                 -- turnDelay는 위 루프에서 갱신됨
                for _, l in ipairs(currentTurnLines) do
                    table.insert(logLines, string.format('<div class="%s" style="animation-delay: %.2fs">%s</div>', l.class, turnDelay, l.text))
                end
                turnDelay = turnDelay + turnInterval
            end
            
            -- 전체 애니메이션 시간 계산 (마지막 턴 기준)
            local totalAnimTime = turnDelay
            
            -- ========== LLM 나레이션 요청 ==========
            local narrationDelay = turnDelay -- 로그 끝난 후에 나레이션 표시
            local narrationHtml = ""
            
            -- 프롬프트 데이터 준비
            local userJson = getLoreBookContent(triggerId, "user") or "{}"
            local userName = json.decode(userJson)["이름"] or summary.userName or "마왕"
            local dungeonLv = summary.dungeonLv or getChatVar(triggerId, "dungeonLv") or "1"
            
            -- summary에서 직접 가져옴 (battle.sys에서 이미 생성됨)
            local bestiaryText = summary.bestiary or "(없음)"
            local heroListText = summary.heroList or (summary.heroCount .. "명의 용사 침공")
            local logText = summary.logText or table.concat(summary.logs, "\n")
            
            -- 결과 요약
            local resultSummary = ""
            if summary.result == "Win" then
                resultSummary = "방어 성공 - 침입자 격퇴"
            elseif summary.result == "BadEnding" then
                resultSummary = "방어 실패 - 마왕 패배"
            else
                resultSummary = summary.result or "미정"
            end
            
            -- 보스전 도달 여부 확인 및 User 정보 포함
            local bossReached = false
            if summary.timeline then
                for _, event in ipairs(summary.timeline) do
                    if event.defender and event.defender.type == "boss" then
                        bossReached = true
                        break
                    end
                end
            end
            
            -- 프롬프트 빌딩
            local battlePtTemplate = getLoreBookContent(triggerId, "battle.pt") or ""
            -- 템플릿 변수 치환
            battlePtTemplate = battlePtTemplate:gsub("{{heroCount}}", tostring(summary.heroCount or 1))
            battlePtTemplate = battlePtTemplate:gsub("{{dungeonLv}}", dungeonLv)
            battlePtTemplate = battlePtTemplate:gsub("{{userName}}", userName)
            battlePtTemplate = battlePtTemplate:gsub("{{heroList}}", heroListText)
            battlePtTemplate = battlePtTemplate:gsub("{{bestiary}}", bestiaryText)
            battlePtTemplate = battlePtTemplate:gsub("{{logText}}", logText)
            battlePtTemplate = battlePtTemplate:gsub("{{resultSummary}}", resultSummary)
            
            local prompt = {
                promptBuild("system", getLoreBookContent(triggerId, "worldBriefing.pt") or ""),
                promptBuild("system", battlePtTemplate)
            }
            
            -- 보스전 도달 시 User 정보 추가
            if bossReached then
                table.insert(prompt, promptBuild("system", "- [마왕 정보] " .. userJson))
            end
            
            -- LLM 호출
            -- setChatVar(triggerId, "battleLoadingState", "전투보고서 작성중...")
            -- reloadDisplay(triggerId)
            local narrationResponse = LLM(triggerId, prompt, "battleNarration")
            
            if narrationResponse and narrationResponse.success then
                local narration = narrationResponse.result or ""
                narration = narration:gsub("<Thoughts>.-</Thoughts>", "") -- CoT 제거
                narrationHtml = string.format('<div class="battle-narration" style="animation-delay: %.2fs">%s</div>', narrationDelay, narration)
                totalAnimTime = narrationDelay + 2.0 -- 나레이션 표시 시간 추가
            else
                local errorMsg = (narrationResponse and narrationResponse.result) or "알 수 없는 오류"
                narrationHtml = string.format('<div class="battle-narration error" style="animation-delay: %.2fs">⚠️ 나레이션 생성 실패: %s</div>', narrationDelay, errorMsg)
            end
            
            -- 저장용 순수 로그 (나레이션 제외)
            setChatVar(triggerId, "battleLogsOnly", table.concat(logLines, "\n"))
            
            table.insert(logLines, narrationHtml)
            
            -- 전체 애니메이션 시간 (로그 + 나레이션)
            local duration = totalAnimTime + 1.0
            
            setChatVar(triggerId, "battleLogsHtml", table.concat(logLines, "\n"))
            setChatVar(triggerId, "battleNarration", narrationHtml:gsub('style="animation%-delay:[^"]*"', '')) -- 애니메이션 제거한 버전
            setChatVar(triggerId, "battleResult", summary.result)
            setChatVar(triggerId, "currentHeroName", "침공 부대 (" .. summary.heroCount .. "명)")
            setChatVar(triggerId, "animDuration", string.format("%.1fs", duration))
            -- 흔들림 효과는 로그 출력 중에만 발생 (나레이션 시간 제외)
            -- turnDelay(narrationDelay)가 마지막 로그의 시작 시간 + interval이므로, 이를 기준으로 설정
            setChatVar(triggerId, "shakeCount", math.ceil(narrationDelay)) 
            
            -- 로딩 완료 - 오버레이 숨김
            setChatVar(triggerId, "battleLoadingState", "")
            reloadDisplay(triggerId)
            debug("전투 화면 로딩 완료")
        else
            debug("⚠️ 전투 결과가 없거나 로그가 비어있습니다.")
            alertNormal(triggerId, "던전 침입이 감지되었으나, 흔적을 찾을 수 없습니다.")
        end
    end
    
    -- 침공이 없을 때만 알림 메시지 표시
    if not invasionTriggered then
        if isNight then
            alertNormal(triggerId, "하루가 지나갑니다...")
        else
            alertNormal(triggerId, "밤이 깊어옵니다...")
        end
    end

    --모든 캐릭터에 대해 회복 처리 (유틸리티 함수 사용)
    for _, charName in ipairs(chars) do
        processRecovery(triggerId, charName, isNight)
    end

    --시간흐름 처리
    day = day + ampm
    ampm = math.abs(ampm - 1)
    debug(day.."일 "..ampm)
    setChatVar(triggerId, "day", day)
    setChatVar(triggerId, "ampm", ampm)

    --갱신된 유저정보 변수저장
    local user = json.decode(getLoreBookContent(triggerId, "user"))
    stateToVar(triggerId, "user", user)
end

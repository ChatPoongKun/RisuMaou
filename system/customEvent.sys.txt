--커스텀 이벤트 실행 엔진
--캐릭터가 보유한 trait/abl/mark/stat/skills/buff의 해당 이벤트를 실행
--@param eventName: 이벤트 이름 (onDCCalc, onStatEXP 등)
--@param context: 이벤트에 전달할 변수 테이블 (must include: tags, target, stat)
--@return context: 수정된 변수 테이블
function(triggerId, eventName, context)
    
    -------------------------------------------------------
    -- Helper Functions
    -------------------------------------------------------
    
    --이벤트 함수 실행 (문자열 or 함수)
    local function executeEventFunc(funcOrStr, ctx)
        if type(funcOrStr) == "function" then
            local status, result = pcall(funcOrStr, ctx)
            if not status then debug("customEvent V2 실행 에러: " .. tostring(result)) end
            return ctx
        end

        local funcStr = funcOrStr
        local env = {}
        for k, v in pairs(ctx) do env[k] = v end
        setmetatable(env, {__index = _G})
        
        local func, err = load(funcStr, "eventFunc", "t", env)
        if func then
            local status, result = pcall(func)
            if status then
                for k, _ in pairs(ctx) do if env[k] ~= nil then ctx[k] = env[k] end end
            else
                debug("customEvent 레거시 실행 에러: " .. tostring(result))
            end
        else
            debug("customEvent 레거시 함수 로드 에러: " .. (err or "unknown"))
        end
        return ctx
    end

    --태그 매칭 로직
    local function tagMatch(tags, hasTags, notTags)
        if hasTags then
            for _, requiredTag in ipairs(hasTags) do
                local found = false
                for _, tag in ipairs(tags) do
                    if tag == requiredTag then found = true; break end
                end
                if not found then return false end
            end
        end
        if notTags then
            for _, excludeTag in ipairs(notTags) do
                for _, tag in ipairs(tags) do
                    if tag == excludeTag then return false end
                end
            end
        end
        return true
    end

    -------------------------------------------------------
    -- Global Cache Logic (Lazy Init)
    -------------------------------------------------------
    if not _G.CustomEventCache then
        debug("[CustomEvent] 캐시 초기화 시작...")
        _G.CustomEventCache = {}
        
        -- 등록할 DB 목록
        local dbList = {
            {name="trait", file="trait.db"},
            {name="abl", file="abl.db"},
            {name="mark", file="mark.db"},
            {name="stat", file="stat.db"},
            {name="skill", file="skills.db"},
            {name="buff", file="buff.db"}
        }

        for _, dbInfo in ipairs(dbList) do
            local dbData = getDB(triggerId, dbInfo.file)
            if dbData then
                for key, def in pairs(dbData) do
                    -- events 필드가 있는 경우에만 등록
                    if def.events then
                        local entryName = def.name or key
                        for evtKey, evtDef in pairs(def.events) do
                            -- 이벤트 이름 추출 (onDCCalc_1 -> onDCCalc)
                            local realEventName = evtKey:match("^([^_]+)") or evtKey
                            
                            if not _G.CustomEventCache[realEventName] then
                                _G.CustomEventCache[realEventName] = {}
                            end

                            table.insert(_G.CustomEventCache[realEventName], {
                                type = dbInfo.name,
                                key = entryName, -- 한글 이름 or 키
                                rawKey = evtKey, -- 실제 이벤트 키 (onDCCalc_2 등)
                                def = evtDef,
                                sourceName = entryName .. "(" .. dbInfo.name .. ")"
                            })
                        end
                    end
                end
            end
        end
        debug("[CustomEvent] 캐시 초기화 완료.")
    end

    -------------------------------------------------------
    -- Main Logic (Cache Based)
    -------------------------------------------------------
    local listeners = _G.CustomEventCache[eventName]
    if not listeners then return context end -- 해당 이벤트 리스너가 없으면 바로 종료

    local target = context.target or getState(triggerId, "target")
    local currentStat = context.stat or getState(triggerId, "stat")
    local tags = context.tags or {}

    for _, listener in ipairs(listeners) do
        -- 보유 여부 확인
        local level = 0
        local hasItem = false
        
        if listener.type == "stat" then
            -- stat은 currentStat 테이블에서 확인
            local val = currentStat[listener.key]
            level = tonumber(val) or 0
            if level > 0 then hasItem = true end
        else
            -- trait, abl, mark, skill, buff는 target 테이블에서 확인
            local val = target[listener.key]
            if val then
                level = tonumber(val) or 1 -- 숫자가 아니면 1 취급 (trait 등)
                if level > 0 then hasItem = true end
            end
        end

        if hasItem then
            -- 정의된 이벤트 실행 (배열 지원)
            local evt = listener.def
            local actionList = {}
            if evt[1] then actionList = evt else table.insert(actionList, evt) end

            for _, action in ipairs(actionList) do
                -- 태그 조건
                local matched = tagMatch(tags, action.hasTags, action.notTags)
                
                -- 타겟 스탯 조건
                if matched and action.targetStat and context.statName then
                    local statMatched = false
                    for _, tStat in ipairs(action.targetStat) do
                        if tStat == context.statName then statMatched = true; break end
                    end
                    if not statMatched then matched = false end
                end

                if matched then
                    -- debug(string.format("[Event:Trigger] %s -> %s", listener.sourceName, eventName))
                    context.level = level
                    context = executeEventFunc(action.func, context)
                end
            end
        end
    end

    return context
end

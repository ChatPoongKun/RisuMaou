-- battle.sys.txt
-- Dungeon Battle System Module

(function()
    ---------------------------------------
    -- ★ 밸런싱 상수 정의 (Balancing Constants)
    -- 전투 시스템의 핵심 수치를 조정하려면 이 섹션의 값을 수정하세요.
    ---------------------------------------
    
    -- [확률 상수] 기력(SP) 0 이하 시 발생하는 이벤트 확률
    local PROB_CAPTURE      = 0.1   -- 나포 확률 (10%). 용사가 기력 소진 시 사로잡힐 확률.
    local PROB_FLEE_SUCCESS = 0.1   -- 도주 성공 확률 (10%). 용사가 기력 소진 시 도망칠 확률. (나머지 80%는 전투 지속)
    
    -- [전투 판정 상수]
    local CRIT_THRESHOLD    = 0.9   -- 치명타 판정 기준. 공격 데미지가 (최대 공격력 * 이 값) 이상이면 치명타로 간주.
    local SP_DMG_MIN        = 5     -- 최소 기력(SP) 피해량. 몬스터 공격 시 용사 기력 감소의 최솟값. (레이팅 차이보다 이 값이 클 경우 적용)
    
    -- [용사 퇴각 조건]
    local RETREAT_HP_RATIO  = 0.3   -- HP 퇴각 기준. 용사 HP가 (최대 HP * 이 값) 미만이면 부상으로 도망.
    local RETREAT_SP_RATIO  = 0.2   -- SP 퇴각 기준. 용사 SP가 (최대 SP * 이 값) 미만이면 공포로 도망.
    
    -- [침공 참여 조건]
    local INVADER_HP_RATIO  = 0.8   -- 침공 참여 HP 조건. 용사 HP가 (최대 HP * 이 값) 이상이어야 침공에 참여.
    local INVADER_SP_RATIO  = 0.8   -- 침공 참여 SP 조건. 용사 SP가 (최대 SP * 이 값) 이상이어야 침공에 참여.
    
    ---------------------------------------
    -- 로컬 함수 정의
    ---------------------------------------

    -- 로그 추가 헬퍼
    local function addLog(battleState, msg)
        table.insert(battleState.logs, msg)
    end

    -- 데미지 계산 헬퍼
    local function calcDamage(atkMin, atkMax, def)
        local atk = math.random(atkMin, atkMax)
        local defense = math.random(0, def)
        local dmg = math.max(0, atk - defense)
        
        local isCrit = false
        if dmg >= (atkMax * CRIT_THRESHOLD) then isCrit = true end
        return dmg, isCrit
    end

    -- 공격력 파싱 헬퍼 (Min~Max or Single)
    local function parseAtk(val)
        if type(val) == "string" and string.find(val, "~") then
            local min, max = val:match("(%d+)~(%d+)")
            return tonumber(min), tonumber(max)
        else
            local v = tonumber(val) or 10
            return v, v
        end
    end

    -- 시너지 로드 (placeholder)
    -- 시너지 로드
    local function loadSynergyModifiers(triggerId)
        local jsonStr = getChatVar(triggerId, "dungeon_modifiers")
        if jsonStr and jsonStr ~= "" then
            -- json 라이브러리는 전역에 있다고 가정 (battle.sys 다른곳에서 씀)
            local status, res = pcall(json.decode, jsonStr)
            if status then return res end
        end
        return {}
    end

    -- 개별 몬스터 전투 루프
    local function runCombatLoop(battleState, monster, slotId)
        local hero = battleState.currentHero
        local floor = battleState.currentFloor
        local slotNum = battleState.currentSlot
        local prefix = string.format("%dF-%d", floor, slotNum)
        
        local monBase = monster.data
        local monMod = battleState.modifiers[slotId] or {}
        
        local function applyMod(base, modVal)
            if not modVal then return base end
            if type(modVal) == "table" then
                local add = modVal.add or 0
                local mul = modVal.mul or 1.0
                return math.floor((base + add) * mul)
            elseif type(modVal) == "number" then
                return base + modVal
            end
            return base
        end

        local monStats = {
            name = monBase.name,
            hp = monster.hp, 
            maxHp = monster.maxHp,
            atkMin = applyMod((monBase.atkMin or 0), monMod.atkMin),
            atkMax = applyMod((monBase.atkMax or 0), monMod.atkMax),
            def = applyMod((monBase.def or 0), monMod.def),
            rating = monBase.rating or 0
        }
        
        addLog(battleState, string.format("%s%s와(과) 조우! (HP: %d/%d, Atk: %d~%d, Def: %d)", 
            prefix, monStats.name, monStats.hp, monStats.maxHp, monStats.atkMin, monStats.atkMax, monStats.def))
            
        local turn = 1
        local loopResult = nil -- "Win", "Lose", "Retreat", "Captured"
        
        while monStats.hp > 0 and hero.hp > 0 do
            local heroFirst = (math.random() < 0.5)
            local order = {}
            if heroFirst then order = {"hero", "monster"} else order = {"monster", "hero"} end
            
            for _, attackerType in ipairs(order) do
                if hero.hp <= 0 or monStats.hp <= 0 then break end
                
                if attackerType == "hero" then
                    local dmg, crit = calcDamage(hero.atkMin, hero.atkMax, monStats.def)
                    local desc = "공격"
                    if crit then desc = "치명타 공격" end
                    if dmg == 0 then desc = "공격이 막힘" end
                    
                    monStats.hp = monStats.hp - dmg
                    addLog(battleState, string.format("[Turn %d] %s 용사의 %s! %s에게 %d 피해. (남은 HP: %d)", 
                        turn, prefix, desc, monStats.name, dmg, math.max(0, monStats.hp)))
                else
                    local dmg, crit = calcDamage(monStats.atkMin, monStats.atkMax, hero.def)
                    local desc = "공격"
                    if crit then desc = "치명타 공격" end
                    if dmg == 0 then desc = "공격이 빗나감" end
                    
                    hero.hp = hero.hp - dmg
                    
                    if dmg > 0 then
                        local spDmg = math.max(SP_DMG_MIN, monStats.rating - (hero.rating or 0))
                        hero.sp = hero.sp - spDmg
                        addLog(battleState, string.format("[Turn %d] %s %s의 %s! 용사에게 %d 피해(기력 %d 감소). (남은 HP: %d, SP: %d)", 
                            turn, prefix, monStats.name, desc, dmg, spDmg, math.max(0, hero.hp), math.max(0, hero.sp)))
                        
                        -- 기력 부족 시 도주/전투/나포 판정
                        if hero.sp <= 0 then
                             local roll = math.random()
                             
                             if roll < PROB_CAPTURE then
                                 -- 10% 나포
                                 addLog(battleState, string.format("[Turn %d] %s 용사가 정신을 잃고 쓰러졌습니다! (나포됨)", turn, prefix))
                                 loopResult = "Captured"
                             elseif roll < (PROB_CAPTURE + PROB_FLEE_SUCCESS) then
                                 -- 10% 도주 성공
                                 addLog(battleState, string.format("[Turn %d] %s 용사가 공포에 질려 도망칩니다!", turn, prefix))
                                 loopResult = "Retreat"
                             else
                                 -- 80% 전투 지속
                                 addLog(battleState, string.format("[Turn %d] %s 용사가 공포에 질렸지만 이 악물고 버팁니다!", turn, prefix))
                                 -- continue
                             end
                        end
                    else
                        -- 데미지 0일때
                        addLog(battleState, string.format("[Turn %d] %s %s의 %s! 용사에게 %d 피해. (남은 HP: %d)", 
                             turn, prefix, monStats.name, desc, dmg, math.max(0, hero.hp)))
                    end
                end
                
                if loopResult then break end
            end
            
            if loopResult then return loopResult, turn end
            
            turn = turn + 1
        end
        
        if monStats.hp <= 0 then
            addLog(battleState, string.format("%s %s 처치!", prefix, monStats.name))
            return "Win", turn
        elseif hero.hp <= 0 then
            return "Lose", turn
        end
        return nil, turn
    end

    -- 보스전 루프
    local function runBossFight(battleState, hero, userStats)
        addLog(battleState, "=== [BOSS BATTLE] 던전 최심부, 마왕의 알현실 ===")
        addLog(battleState, string.format("마왕 %s(이)가 침입자 %s(을)를 맞이합니다!", battleState.userName, hero.name))
        
        -- 마왕 공격력 파싱
        -- 마왕 공격력 파싱
        local bossAtkMin, bossAtkMax = parseAtk(userStats.atk)

        local bossStats = {
            name = battleState.userName,
            hp = userStats.hp, maxHp = userStats.maxHp,
            atkMin = bossAtkMin, atkMax = bossAtkMax, def = userStats.def
        }
        
        local turn = 1
        while bossStats.hp > 0 and hero.hp > 0 and hero.sp > 0 do
            local dmg, crit = calcDamage(bossStats.atkMin, bossStats.atkMax, hero.def)
            hero.hp = hero.hp - dmg
            addLog(battleState, string.format("[Turn %d] 마왕의 강력한 일격! %s에게 %d 피해. (남은 HP: %d)", turn, hero.name, dmg, math.max(0, hero.hp)))
            
            if hero.hp <= 0 then return "Win", bossStats.hp end
            
            dmg, crit = calcDamage(hero.atkMin, hero.atkMax, bossStats.def)
            bossStats.hp = bossStats.hp - dmg
            addLog(battleState, string.format("[Turn %d] %s의 필사적인 공격! 마왕에게 %d 피해. (남은 HP: %d)", turn, hero.name, dmg, math.max(0, bossStats.hp)))
            
            turn = turn + 1
        end
        
        if bossStats.hp <= 0 then return "Lose", 0 else return "Win", bossStats.hp end
    end

    -- 메인 침공 프로세스
    local function processInvasion(triggerId)
        local dungeon = getState(triggerId, "dungeon") or { floors=1, grid={} }
        local modifiers = loadSynergyModifiers(triggerId)
        local monstersDB = getDB(triggerId, "monsters.db")
        
        -- Safe decode chars
        local charsStr = getChatVar(triggerId, "chars")
        local chars = {}
        if charsStr and charsStr ~= "" then
            local status, res = pcall(json.decode, charsStr)
            if status then chars = res else debug("battle.sys: chars decode failed") end
        end

        -- Safe decode user
        local userStr = getLoreBookContent(triggerId, "user") or "{}"
        local userJson = {}
        if userStr and userStr ~= "" then
            local status, res = pcall(json.decode, userStr)
            if status then userJson = res else debug("battle.sys: user decode failed") end
        end
        local userName = userJson["이름"] or "마왕"
        
        local heroes = {}
        local waitingHeroes = {}
        
        for _, name in ipairs(chars) do
            local charStr = getLoreBookContent(triggerId, name)
            local charData = {}
            if charStr and charStr ~= "" then
                local status, res = pcall(json.decode, charStr)
                if status then charData = res end
            end
            
            local hp = tonumber(charData["체력"]) or 0
            local maxHp = tonumber(charData["최대체력"]) or 100
            local sp = tonumber(charData["기력"]) or 0
            local maxSp = tonumber(charData["최대기력"]) or 100
            
            -- 침공 조건: 정비중 + 체력/기력 조건 충족
            if charData.condition == "REST" and hp >= (maxHp * INVADER_HP_RATIO) and sp >= (maxSp * INVADER_SP_RATIO) then
                table.insert(waitingHeroes, charData)
            end
        end
        
        local requiredHeroes = 1
        local loopCount = 0
        local MAX_LOOP = 5
        
        while #waitingHeroes < requiredHeroes and loopCount < MAX_LOOP do
            loopCount = loopCount + 1
            sysFunction(triggerId, "newChar.sys", "invasion")
            
            charsStr = getChatVar(triggerId, "chars")
            chars = {}
            if charsStr and charsStr ~= "" then
                local status, res = pcall(json.decode, charsStr)
                if status then chars = res end
            end

            local newName = chars[#chars]
            if newName then
                local newCharStr = getLoreBookContent(triggerId, newName)
                local newChar = {}
                if newCharStr and newCharStr ~= "" then
                     local status, res = pcall(json.decode, newCharStr)
                     if status then newChar = res end
                end
                
                local hp = tonumber(newChar["체력"]) or 0
                local maxHp = tonumber(newChar["최대체력"]) or 100
                local sp = tonumber(newChar["기력"]) or 0
                local maxSp = tonumber(newChar["최대기력"]) or 100
                
                if newChar.condition == "REST" and hp >= (maxHp * INVADER_HP_RATIO) and sp >= (maxSp * INVADER_SP_RATIO) then
                   table.insert(waitingHeroes, newChar)
                end
            else
                debug("battle.sys: Failed to get new character name")
            end
        end
        
        -- If strictly required, pick whatever we have or duplicate
        if #waitingHeroes == 0 then
             -- Emergency fallback
             table.insert(waitingHeroes, {["이름"]="용사(임시)", ["체력"]=100, ["최대체력"]=100, ["기력"]=100, ["최대기력"]=100, ["공격력"]=10, ["방어력"]=5, ["레이팅"]=10})
        end

        for i = 1, math.min(#waitingHeroes, requiredHeroes) do
            table.insert(heroes, waitingHeroes[i])
        end
        
        local battleState = {
            heroes = heroes,
            currentHeroIdx = 1,
            currentHero = nil,
            userName = userName,
            modifiers = modifiers,
            logs = {},
            monstersList = {},
            currentFloor = 1,
            currentSlot = 1
        }
        
        addLog(battleState, string.format("🚨 경보! %d명의 용사가 던전을 침공했습니다!", #heroes))
        
        local finalResult = "Ongoing"
        
        for i, h in ipairs(heroes) do
            h.hp = tonumber(h["체력"]) or 100
            h.maxHp = tonumber(h["최대체력"]) or 100
            h.sp = tonumber(h["기력"]) or 100
            h.maxSp = tonumber(h["최대기력"]) or 100
            
            -- 공격력 파싱 (min~max)
            local amin, amax = parseAtk(h["공격력"])
            h.atkMin = amin
            h.atkMax = amax
            
            h.def = tonumber(h["방어력"]) or 10
            h.rating = tonumber(h["레이팅"]) or 10
            h.name = h["이름"]
        end
        
        local clearedSlots = {}
        local timeline = {} -- [NEW] 전투 타임라인 기록
        local globalTurnIndex = 0 -- 로그 턴 배치 순서를 위한 누적 인덱스
        
        while battleState.currentHeroIdx <= #heroes do
            local hero = heroes[battleState.currentHeroIdx]
            battleState.currentHero = hero
            
            addLog(battleState, string.format(">>> %d번째 용사 [%s] (Rating: %d) 출격!", 
                battleState.currentHeroIdx, hero.name, hero.rating))
            
            -- 출격 로그는 턴으로 치지 않거나, 0턴으로 취급
            
            local heroDefeated = false
            
            for f = battleState.currentFloor, dungeon.floors do
                battleState.currentFloor = f
                
                local floorHasUnits = false
                local startSlot = 1
                if f == battleState.currentFloor then startSlot = battleState.currentSlot end
                if startSlot > 5 then startSlot = 1 end
                
                for s = startSlot, 5 do
                    battleState.currentSlot = s
                    local slotId = string.format("%d_%d", f, s)
                    local node = dungeon.grid and dungeon.grid[slotId]
                    
                    -- 타임라인 이벤트 기록 (이동/조우)
                    local currentDefender = { type="none", key="" }
                    
                    if node then
                        floorHasUnits = true
                        if node.type == "monster" then
                            local monData = monstersDB[node.key]
                            if monData then
                                currentDefender = { type="monster", key=node.key }
                                local monster = {
                                    key = node.key, hp = monData.hp, maxHp = monData.hp, data = monData
                                }
                                battleState.monstersList[monData.name] = monData.desc
                                
                                -- 타임라인 저장 (이 조우가 시작되는 누적 턴 인덱스 기록)
                                table.insert(timeline, {
                                    startTurnIndex = globalTurnIndex,
                                    floor = f,
                                    slot = s,
                                    defender = currentDefender,
                                    slotId = slotId
                                })

                                local outcome, combatTurns = runCombatLoop(battleState, monster, slotId)
                                globalTurnIndex = globalTurnIndex + (combatTurns or 1) -- 이 전투에서 사용된 턴만큼 증가
                                
                                if outcome == "Lose" then
                                    addLog(battleState, hero.name.."이(가) 쓰러졌습니다!")
                                    hero["condition"] = "FAINT"
                                    heroDefeated = true
                                    break
                                elseif outcome == "Captured" then
                                    addLog(battleState, hero.name.."이(가) 적에게 사로잡혔습니다!")
                                    hero["condition"] = "TRAINABLE"
                                    heroDefeated = true
                                    break
                                elseif outcome == "Retreat" then
                                    addLog(battleState, hero.name.."이(가) 도망쳤습니다!")
                                    heroDefeated = true
                                    break
                                end
                                
                                -- 승리
                                clearedSlots[slotId] = true
                                
                                if hero.hp < (hero.maxHp * RETREAT_HP_RATIO) then
                                     addLog(battleState, hero.name.."이(가) 부상을 입고 도망칩니다...")
                                     heroDefeated = true
                                     break
                                elseif hero.sp < (hero.maxSp * RETREAT_SP_RATIO) then
                                     addLog(battleState, hero.name.."이(가) 공포에 질려 도망칩니다...")
                                     heroDefeated = true
                                     break
                                end
                            end
                            
                        elseif node.type == "trap" then
                            currentDefender = { type="trap", key=node.key }
                            table.insert(timeline, {
                                startTurnIndex = globalTurnIndex,
                                floor = f,
                                slot = s,
                                defender = currentDefender,
                                slotId = slotId
                            })
                            globalTurnIndex = globalTurnIndex + 1 -- 함정은 1턴만 소모
                            
                            addLog(battleState, string.format("%dF-%d 함정 발동!", f, s))
                            hero.hp = hero.hp - 50
                            clearedSlots[slotId] = true
                            
                            if hero.hp <= 0 then
                                addLog(battleState, hero.name.."이(가) 함정에 걸려 쓰러졌습니다!")
                                hero["condition"] = "FAINT"
                                heroDefeated = true
                                break
                            end
                        end
                    else
                        -- 빈 슬롯
                        clearedSlots[slotId] = true
                        -- 빈 슬롯도 턴(시간)을 소모하는가? 로그에는 "고요합니다"가 층 단위로 뜸.
                        -- 시각적으로는 '지나가는' 모션이 필요할 수 있음. 일단 스킵하거나 빠르게 지나가도록.
                    end
                end
                
                if not floorHasUnits then
                    addLog(battleState, string.format("%dF는 고요합니다... 용사가 발걸음을 재촉합니다.", f))
                end
                
                if heroDefeated then break end
                 battleState.currentSlot = 1 
            end
            
            if not heroDefeated then
                table.insert(timeline, {
                    startTurnIndex = globalTurnIndex,
                    floor = 99, -- Boss Room
                    slot = 99,
                    defender = { type="boss", key="user" },
                    slotId = "boss"
                })
                
                addLog(battleState, hero.name.."이(가) 마왕의 방에 도달했습니다!")
                local userStats = {
                    hp = tonumber(userJson["체력"]) or 1000,
                    maxHp = tonumber(userJson["최대체력"]) or 1000,
                    atk = tonumber(userJson["공격력"]) or 100,
                    def = tonumber(userJson["방어력"]) or 100,
                }
                
                local bossResult, remainingHp = runBossFight(battleState, hero, userStats)
                
                -- 유저 HP 업데이트 및 저장
                userJson["체력"] = math.max(0, remainingHp)
                upsertLocalLoreBook(triggerId, "user", json.encode(userJson), DEFAULT_LORE_OPTION)
                
                if bossResult == "Lose" then
                    finalResult = "BadEnding"
                    break
                else
                    addLog(battleState, "마왕이 용사를 직접 제압했습니다!")
                    hero["condition"] = "FAINT"
                    heroDefeated = true
                end
            end
            
            -- HP/SP 변화를 로어북 키에 반영
            hero["체력"] = math.max(0, hero.hp)
            hero["기력"] = math.max(0, hero.sp)
            
            upsertLocalLoreBook(triggerId, hero.name, json.encode(hero), DEFAULT_LORE_OPTION)
            
            battleState.currentHeroIdx = battleState.currentHeroIdx + 1
        end
        
        if finalResult == "BadEnding" then
            addLog(battleState, "마왕성이 함락되었습니다...")
        else
            addLog(battleState, "방어에 성공했습니다.")
            finalResult = "Win"
        end
        
        local bestiary = {}
        for name, desc in pairs(battleState.monstersList) do
            table.insert(bestiary, string.format("- %s: %s", name, desc))
        end
        
        local heroList = {}
        for _, h in ipairs(heroes) do
            table.insert(heroList, string.format("- %s (HP:%s/Rating:%s)", h.name, h.maxHp, h.rating))
        end
        
        local cleanLogs = {}
        for _, line in ipairs(battleState.logs) do
            table.insert(cleanLogs, line)
        end
        local logText = table.concat(cleanLogs, "\n")
        
        local summary = {
            result = finalResult,
            logs = battleState.logs,
            logText = logText,
            heroList = table.concat(heroList, "\n"),
            bestiary = table.concat(bestiary, "\n"),
            heroCount = #heroes,
            dungeonLv = dungeon.floors,
            userName = userName,
            clearedSlots = clearedSlots,

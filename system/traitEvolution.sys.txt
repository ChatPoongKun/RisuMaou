-- traitEvolution.sys
-- 특성 진화 및 획득 엔진
-- traitRules.sys를 로드하여 조건을 검사하고, 새로운 특성을 부여합니다.

function(triggerId, target)
    debug("traitEvolution 실행")
    
    local traitDB = getDB(triggerId, "trait.db")
    
    -- 규칙 파일 로드 (pcall로 안전하게)
    local rules_ok, rules = pcall(function() 
        return sysFunction(triggerId, "traitRules.sys") 
    end)
    
    if not rules_ok or type(rules) ~= "table" then
        debug("traitRules 로드 실패 또는 형식이 잘못됨: " .. tostring(rules))
        return target, ""
    end
    
    -- 필요한 데이터 로드
    local stats = getState(triggerId, "stat")
    local exps = getState(triggerId, "target") -- target 자체가 경험치 필드를 포함하고 있음 (평탄화된 구조)
    -- 주의: target은 로어북에서 온 평탄화된 데이터. 경험치 키값(한글)이 포함되어 있는지 확인 필요.
    -- newChar.sys 등을 보면 target["V경험"] 처럼 직접 저장됨. (exp.db의 name 필드가 키로 사용됨)
    -- 따라서 target을 그대로 exps로 사용해도 무방.
    
    local newTraits = {}
    local logs = ""
    
    for key, checkFn in pairs(rules) do
        -- DB에 존재하는 특성인지 확인
        local traitDef = traitDB[key]
        if traitDef then
            local traitName = traitDef.name
            
            -- 이미 보유 중인지 확인 
            -- (target[traitName] == 1 또는 "1" 등. 확실하게 체크하기 위해 tonumber 사용)
            local currentVal = tonumber(target[traitName]) or 0
            
            if currentVal == 0 then
                -- 조건 검사
                local success, result = pcall(checkFn, triggerId, target, stats, target)
                
                if success and result == true then
                    -- 특성 부여
                    target[traitName] = 1
                    table.insert(newTraits, traitName)
                    debug("새로운 특성 획득: " .. traitName)
                    
                    -- 상충되는 특성 제거 로직 (예: 민감 <-> 둔감)
                    -- traitDef에 "opposite" 필드가 있다면 좋겠지만, 지금은 하드코딩하거나 나중에 DB확장 필요.
                    -- 간단하게 이름 기반으로 추론해서 제거 (예: V민감 <-> V둔감)
                    if traitName:find("민감") then
                        local dullName = traitName:gsub("민감", "둔감")
                        if target[dullName] and tonumber(target[dullName]) > 0 then
                            target[dullName] = 0
                            debug("상충 특성 제거: " .. dullName)
                        end
                    elseif traitName:find("명기") then
                        -- 명기가 되면 둔감은 확실히 제거
                        local dullName = traitName:gsub("명기", "둔감") -- V명기 -> V둔감
                        if target[dullName] and tonumber(target[dullName]) > 0 then
                            target[dullName] = 0
                        end
                    end
                    
                    -- 멋진 연출을 위한 HTML 구조
                    logs = logs .. [[
<div class='getTrait-overlay' style='position:fixed; top:50%; left:50%; transform:translate(-50%, -50%); z-index:9999; animation: popIn 0.5s cubic-bezier(0.175, 0.885, 0.32, 1.275); pointer-events:none;'>
<div class='getTrait-box' style='background: rgba(0,0,0,0.8); padding: 20px; border: 2px solid gold; border-radius: 10px; color: white; text-align: center; box-shadow: 0 0 20px gold;'>
<div style='font-size: 1.2em; color: gold; margin-bottom: 5px;'>✨ 특성 개화 ✨</div>
<div style='font-size: 1.5em; font-weight: bold;'>]] .. traitName .. [[</div>
<div style='font-size: 0.8em; color: #ccc; margin-top: 5px;'>조건 충족으로 인해 새로운 특성을 획득했습니다.</div>
</div>
</div>
]]
                elseif not success then
                    debug("Rule Check Error [" .. key .. "]: " .. tostring(result))
                end
            end
        end
    end
    
    return target, logs
end

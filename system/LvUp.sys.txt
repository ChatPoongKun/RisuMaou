-- LvUp.sys.txt
-- 캐릭터 레벨업 처리 모듈

(function()
    ---------------------------------------
    -- ★ 밸런싱 상수 정의 (Balancing Constants)
    ---------------------------------------

    -- [경험치 테이블 오프셋]
    local EXP_LEVEL_OFFSET = 3          -- 레벨 N 도달에 필요한 경험치 = EXPtable[N + 이 값]

    -- [경험치 테이블 오프셋]
    local EXP_LEVEL_OFFSET = 3          -- 레벨 N 도달에 필요한 경험치 = EXPtable[N + 이 값]

    ---------------------------------------

    local funcs = {}

    local function getClassData(triggerId, className)
        local classDB = getDB(triggerId, "class.db")
        for key, data in pairs(classDB) do
            if data.name == className then return data end
        end
        return { hp_inc=100, sp_inc=100, def_inc=5, atk_min=3, atk_max=6 }
    end

    function funcs.calcRating(triggerId, target)
        local hp = tonumber(target["최대체력"]) or 0
        local def = tonumber(target["방어력"]) or 0
        local atkStr = target["공격력"] or "0~0"
        local atkMin, atkMax = atkStr:match("(%d+)~(%d+)")
        atkMin = tonumber(atkMin) or 0
        atkMax = tonumber(atkMax) or 0
        
        -- [밸런싱 DB 로드]
        local balanceDB = getDB(triggerId, "balancing.db")
        local rConfig = balanceDB.rating or { hp=0.3, def=4.5, atkMin=4.0, atkMax=2.7 }
        
        local rating = math.floor(
            hp * (rConfig.hp or 0.3) + 
            def * (rConfig.def or 4.5) + 
            atkMin * (rConfig.atkMin or 4.0) +
            atkMax * (rConfig.atkMax or 2.7)
        )
        target["레이팅"] = rating
        return target
    end

    function funcs.check(triggerId, target)
        local expTable = getDB(triggerId, "EXPtable.db")
        local currentLevel = tonumber(target["레벨"]) or 1
        local currentEXP = tonumber(target["경험치"]) or 0
        
        local reqKey = tostring(currentLevel + EXP_LEVEL_OFFSET)
        local reqEXP = expTable[reqKey]
        
        if not reqEXP then return false, 0, 0 end
        return currentEXP >= reqEXP, currentEXP, reqEXP
    end

    function funcs.execute(triggerId, target, count, silent)
        count = count or 1
        silent = silent or false
        
        local classData = getClassData(triggerId, target["클래스"])
        
        local totalHPGain = 0
        local totalSPGain = 0
        local totalAtkMinGain = 0
        local totalAtkMaxGain = 0
        local totalDefGain = 0
        local levelsGained = 0
        
        for i = 1, count do
            local hpGain = math.random(1, classData.hp_inc or 100)
            local spGain = math.random(1, classData.sp_inc or 100)
            
            -- [Round] HP/SP 10단위 반올림
            hpGain = math.floor((hpGain + 5) / 10) * 10
            spGain = math.floor((spGain + 5) / 10) * 10
            
            -- 최소값 보정 (0이 되면 섭섭하니 10으로)
            if hpGain < 10 then hpGain = 10 end
            if spGain < 10 then spGain = 10 end
            local defGain = math.random(1, classData.def_inc or 5)
            local atkMinGain = math.random(1, classData.atk_min or 3)
            local atkMaxGain = math.random(1, classData.atk_max or 6)
            
            local maxHP = tonumber(target["최대체력"]) or 100
            local maxSP = tonumber(target["최대기력"]) or 100
            local currentHP = tonumber(target["체력"]) or maxHP
            local currentSP = tonumber(target["기력"]) or maxSP
            local def = tonumber(target["방어력"]) or 0
            
            local atkStr = target["공격력"] or "0~0"
            local atkMin, atkMax = atkStr:match("(%d+)~(%d+)")
            atkMin = tonumber(atkMin) or 0
            atkMax = tonumber(atkMax) or 0
            
            maxHP = maxHP + hpGain
            maxSP = maxSP + spGain
            currentHP = currentHP + hpGain
            currentSP = currentSP + spGain
            atkMin = atkMin + atkMinGain
            atkMax = atkMax + atkMaxGain
            
            -- [안전장치] min > max 역전 시 스왑
            if atkMax < atkMin then
                local temp = atkMin
                atkMin = atkMax
                atkMax = temp
            end
            
            def = def + defGain
            
            target["최대체력"] = tostring(maxHP)
            target["최대기력"] = tostring(maxSP)
            target["체력"] = tostring(math.min(currentHP, maxHP))
            target["기력"] = tostring(math.min(currentSP, maxSP))
            target["공격력"] = atkMin .. "~" .. atkMax
            target["방어력"] = def
            
            local currentLevel = tonumber(target["레벨"]) or 1
            target["레벨"] = tostring(currentLevel + 1)
            
            totalHPGain = totalHPGain + hpGain
            totalSPGain = totalSPGain + spGain
            totalAtkMinGain = totalAtkMinGain + atkMinGain
            totalAtkMaxGain = totalAtkMaxGain + atkMaxGain
            totalDefGain = totalDefGain + defGain
            levelsGained = levelsGained + 1
        end
        
        target = funcs.calcRating(triggerId, target)
        
        local screenEffect = ""
        if not silent and levelsGained > 0 then
            local name = target["이름"] or "대상"
            local newLevel = target["레벨"]
            
            screenEffect = string.format([[
<div style="position:fixed; top:50%%; left:50%%; transform:translate(-50%%, -50%%); z-index:9999; 
background:rgba(0,0,0,0.8); border:2px solid #60d9f7; border-radius:10px; padding:20px; 
color:#fff; text-align:center; min-width:300px; box-shadow:0 0 20px rgba(96,217,247,0.5);
animation: fadeInOut 4s forwards;">
    <div style="font-size:1.5rem; font-weight:bold; color:#60d9f7; margin-bottom:10px;">LEVEL UP!</div>
    <div style="font-size:1.1rem; color:#d4af37; margin-bottom:15px;">%s (Lv.%s)</div>
    <div style="display:grid; grid-template-columns: 1fr 1fr; gap:10px; text-align:left; font-size:0.9rem;">
        <div>HP <span style="color:#d32f2f;">+%d</span></div>
        <div>SP <span style="color:#1976d2;">+%d</span></div>
        <div>ATK <span style="color:#e64a19;">+%d~%d</span></div>
        <div>DEF <span style="color:#388e3c;">+%d</span></div>
    </div>
</div>
]], name, newLevel, totalHPGain, totalSPGain, totalAtkMinGain, totalAtkMaxGain, totalDefGain)
            
            debug("[LvUp] " .. name .. ": Lv " .. newLevel .. " HP+" .. totalHPGain)
        end
        
        return target, screenEffect, levelsGained
    end

    return function(triggerId, subFunc, ...)
        if funcs[subFunc] then return funcs[subFunc](triggerId, ...) end
        debug("LvUp.sys: Unknown function " .. tostring(subFunc))
    end
end)()

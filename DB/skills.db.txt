-- 5. Weak Link (약점 보완)
weak_link = {
    name = "약점 보완",
    desc = "자신 방어력이 낮고(5↓) 주변 방어력이 높을 때(20↑) 방어력 증가",
    active = function(ctx)
        local selfUnit = ctx.grid[ctx.slotId]
        if not selfUnit or not selfUnit.data then return {} end
        
        local selfDef = tonumber(selfUnit.data.def) or 0
        if selfDef <= 5 then
            local neighbors = ctx:neighbors()
            if neighbors:count() == 2 then
                local conditionMet = true
                for _, item in ipairs(neighbors.units) do
                     local def = (item.unit.data and tonumber(item.unit.data.def)) or 0
                     if def < 20 then conditionMet = false break end
                end
                
                if conditionMet then
                    return {{ target = ctx.slotId, buff = "weak_link_boost", source = ctx.slotId }}
                end
            end
        end
        return {}
    end
}

-- 9. Undead March (망자의 행진)

    name = "망자의 행진",

    active = function(ctx)
        local f, s = ctx.slotId:match("^(%d+)_(%d+)$")
        s = tonumber(s)
        
        local count = 0
        for i = 1, s-1 do
            local prevKey = string.format("%d_%d", f, i)
            local unit = ctx.grid[prevKey]
            if unit and unit.hasTag and unit:hasTag("언데드") then count = count + 1 end
        end
        
        if count > 0 then
             return {{ target = ctx.slotId, buff = "undead_march_buff", source = ctx.slotId }} 
        end
        return {}
    end
}

-- 19. Beast Howl (야수의 포효)
beast_howl = {
    name = "야수의 포효",
    desc = "중앙(3번) 배치 시 좌우 아군 강화",
    active = function(ctx)
        local f, s = ctx.slotId:match("^(%d+)_(%d+)$")
        s = tonumber(s)
        
        if s == 3 and ctx.owner:hasTag("야수") then
            local targets = {}
            local left = ctx.grid[string.format("%d_%d", f, 2)]
            local right = ctx.grid[string.format("%d_%d", f, 4)]
            
            if left then table.insert(targets, { target=string.format("%d_%d", f, 2), buff="beast_howl_buff", source=ctx.slotId }) end
            if right then table.insert(targets, { target=string.format("%d_%d", f, 4), buff="beast_howl_buff", source=ctx.slotId }) end

        end
        return {}
    end

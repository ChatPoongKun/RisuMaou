[charInfo/use_영양제/hidden] function(triggerId)
    local itemName = "영양제"
    local inventory = json.decode(getChatVar(triggerId, "inventory"))
    local target = json.decode(getChatVar(triggerId, "target"))
    local target = json.decode(getLoreBookContent(triggerId, target["이름"]))
    local max = tonumber(target["최대체력"])
    local recover = math.floor(max * 0.25) --최대 체력의 25% 회복

    if tonumber(inventory[itemName] or 0) <= 0 then
        alertNormal(triggerId, itemName.."이 부족합니다.")
        return
    elseif tonumber(target["체력"]) >= max then
        alertNormal(triggerId, target["이름"].."의 체력이 이미 최대치입니다.")
        return
    end

    -- 회복 효과 적용
    target["체력"] = intForm(math.min(max, tonumber(target["체력"]) + recover))

    -- 인벤토리 차감 및 저장
    inventory[itemName] = tostring(tonumber(inventory[itemName]) - 1)
    setChatVar(triggerId, "inventory", json.encode(inventory))

    -- 타겟 정보 저장
    local option = {alwaysActive = false, insertOrder = 100, key = "", secondKey = "", regex = false}
    upsertLocalLoreBook(triggerId, target["이름"], json.encode(target), option)
    setState(triggerId, "target", target)
    stateToVar(triggerId, "target", target)

    alertNormal(triggerId, target["이름"].."에게 "..itemName.."를 사용했습니다. ("..target["체력"].."/"..max..")")
end!!

[charInfo/use_강모제/hidden] function(triggerId)
    local itemName = "강모제"
    local inventory = json.decode(getChatVar(triggerId, "inventory"))
    local target = json.decode(getChatVar(triggerId, "target"))
    local target = json.decode(getLoreBookContent(triggerId, target["이름"]))
    local max = "무성함"

    if tonumber(inventory[itemName] or 0) <= 0 then
        alertNormal(triggerId, itemName.."이 부족합니다.")
        return
    elseif hasVal(target, "무모증") then
        alertNormal(triggerId, target["이름"].."의 음모는 무모증으로 자라날 수 없습니다.")
        return
    end

    -- 효과 적용
    if target["음모"] == "없음" then
        target["음모"] = "짧음"
    elseif target["음모"] == "짧음" then
        target["음모"] = "보통"
    elseif target["음모"] == "보통" then
        target["음모"] = "무성함"
    end

    -- 인벤토리 차감 및 저장
    inventory[itemName] = tostring(tonumber(inventory[itemName]) - 1)
    setChatVar(triggerId, "inventory", json.encode(inventory))

    -- 타겟 정보 저장
    local option = {alwaysActive = false, insertOrder = 100, key = "", secondKey = "", regex = false}
    upsertLocalLoreBook(triggerId, target["이름"], json.encode(target), option)
    setState(triggerId, "target", target)
    stateToVar(triggerId, "target", target)

    alertNormal(triggerId, target["이름"].."의 음모가 자라났습니다.("..target["음모"]..")")
end!!

[charInfo/popup_portrait/hidden] function(triggerId)
    local targetVar = getChatVar(triggerId, "target")
    local target = json.decode(targetVar)
    
    if type(target) == "string" then
       local content = getLoreBookContent(triggerId, target)
       if content and content ~= "" then
           target = json.decode(content)
       end
    elseif target["이름"] then
        local content = getLoreBookContent(triggerId, target["이름"])
        if content and content ~= "" then
            target = json.decode(content)
        end
    end

    if target then
        local pre = target["portrait_pre"] or ""
        local append = target["portrait_append"] or ""
        local imgTag = pre .. append
        setChatVar(triggerId, "charPopupImg", imgTag)
        
        -- inlay 체크
        if string.find(pre, "{{inlay") then
            setChatVar(triggerId, "isInlay", "true")
        else
            setChatVar(triggerId, "isInlay", "")
        end
    end
    setChatVar(triggerId, "showConvertConfirm", "")
end!!

[charInfo/popup_convert_confirm/hidden] function(triggerId)
    local targetVar = getChatVar(triggerId, "target")
    local target = json.decode(targetVar)
    
    if type(target) == "string" then
       local content = getLoreBookContent(triggerId, target)
       if content and content ~= "" then target = json.decode(content) end
    elseif target["이름"] then
        local content = getLoreBookContent(triggerId, target["이름"])
        if content and content ~= "" then target = json.decode(content) end
    end
    
    if not target then return end
    
    local name = target["이름"]
 
    target["portrait_pre"] = "<img src='{{raw:"
    target["portrait_append"] = ":" .. name .. "}}'>"
    
    -- 저장
    local option = {alwaysActive = false, insertOrder = 100, key = "", secondKey = "", regex = false}
    upsertLocalLoreBook(triggerId, name, json.encode(target), option)
    
    -- 현재 상태 업데이트
    setChatVar(triggerId, "charPopupImg", target["portrait_pre"] .. target["portrait_append"])
    setChatVar(triggerId, "isInlay", "")
    setChatVar(triggerId, "showConvertConfirm", "")
    
    setState(triggerId, "target", target)
    stateToVar(triggerId, "target", target)
    
    alertNormal(triggerId, "초상화 타입이 Raw(로컬파일)로 변경되었습니다.")
end!!
